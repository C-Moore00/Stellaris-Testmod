# Remove orbital arcology
decision_remove_orbital_arcology = {
	owned_planets_only = yes
	sound = event_construction
	icon = decision_remove_orbital_arcology
	prerequisites = { "giga_tech_orbital_arcologies" }
	ai_weight = {
		weight = 0
	}

	potential = {
		exists = owner
		or = {
			is_artificial = no
			has_planet_flag = giga_planet_mega
		}
		has_planet_flag = orbital_arcology
		solar_system = {
			any_system_megastructure = {
				or = {
					is_megastructure_type = orbital_arcology_3
					is_megastructure_type = orbital_arcology_2
					is_megastructure_type = orbital_arcology_1
				}
				planet = { is_same_value = ROOT }
				is_upgrading = no
			}
		}
		NOT = {
			has_global_flag = orbital_arcologies_disabled
		}
	}
	
	allow = {
		has_fully_upgraded_capital = yes
	}

	resources = {
		category = giga_megastructures
		cost = {
			energy = 500
			multiplier = trigger:planet_size
		}
	}

	effect = {
		hidden_effect = {
			solar_system = {
				random_system_megastructure = {
					limit = {
						or = {
							is_megastructure_type = orbital_arcology_3
							is_megastructure_type = orbital_arcology_2
							is_megastructure_type = orbital_arcology_1
						}
						planet = { is_same_value = ROOT } 
					}
					remove_megastructure = this
				}
			}
			remove_planet_flag = orbital_arcology_1
			remove_planet_flag = orbital_arcology_2
			remove_planet_flag = orbital_arcology_3
			giga_clear_orbital_arcology_modifiers = yes
			remove_modifier = new_orbital_arcology_triple_modifier
			remove_planet_flag = orbital_arcology
			add_planet_devestation = 20
		}
	}
	enactment_time = 120
}
