# Machine Factory Project
decision_machinefactory_project = {
	owned_planets_only = yes
	icon = decision_machine_factory_project

	enactment_time = 360
	resources = {
		category = decisions
		cost = {
			influence = 200
			minerals = 10000
		}
	}
	
	potential = {
		NOR = {
			is_planet_class = pc_machine_factory
		}
		OR = {
			is_planet_class = pc_machine
			is_planet_class =  pc_machine_cave
		}
		is_artificial = no
		is_pd_habitat = no
		is_pd_planetary_megaproject = no
		exists = owner
		owner = { has_ascension_perk = ap_machine_worlds }
		owner = { not = { has_civic = civic_machine_servitor } }
	}
	
	allow = {
		custom_tooltip = {
			fail_text = decision_arcology_project_requirements
			free_district_slots < 1
			num_districts = { type = district_farming value < 1 }
			num_districts = { type = district_mining value < 1 }
			num_districts = { type = district_generator value < 1 }
			num_districts = { type = district_farming_uncapped value < 1 }
			num_districts = { type = district_mining_uncapped value < 1 }
			num_districts = { type = district_generator_uncapped value < 1 }
		}
		num_uncleared_blockers < 1
	}
	
	effect = {
		# Make city-planet
		custom_tooltip = decision_machine_factory_effects
		hidden_effect = {		
			change_pc = pc_machine_factory
			owner = {
				country_event = { id = distar.201 } # port for living planet distar.197, triggers in event
			}
			planet_event = { id = pdengine.10 }
			planet_event = { id = pdengine.14 }
			add_deposit = d_pd_industrial_sector
			relic_world_conversion_effect = {
				DISTRICT = district_mf_alloy
				VARIABLE = num_industrial_districts
			}
			IF = {
				limit = { 
					has_deposit = d_pd_necro_arc
				}
				IF = { limit = { NOT = { has_deposit = d_pd_machinecap_arc } }
				set_planet_entity = { picture = pc_machine_necropolis entity = machinenecrofactory_planet_01_entity }
				}
				IF = { limit = { has_deposit = d_pd_machinecap_arc }
				set_planet_entity = { picture = pc_machine_necropolis_capital entity = machinenecrofactory_planet_01_entity }
				}
			}
			IF = {
				limit = { has_deposit = d_pd_machinecap_arc }
				set_planet_entity = { picture = pc_machinefort_capital entity = machine_fort_planet_01_entity }
			}
		}
	}

	ai_weight = {
		weight = 1
        modifier = {
            factor = 100
            exists = owner
            num_pops >= 70
        }
		modifier = { 
			factor = 0	
			owner = { any_owned_planet = { is_planet_class = pc_machine_factory } }
		}
	}
}

	icon = decision_terraform
	enactment_time = 0

	resources = {
		category = decisions
		cost = {
		}
	}

	potential = {
		exists = owner
		OR = {
			AND = {
				has_district = xvcv_mdlc_pc_virtual_district_primary
				NAND = {
					owner = { has_ascension_perk = xvcv_mdlc_ap_virtual_worlds }
					is_xvcv_mdlc_pc_virtual = yes
					uses_district_set = xvcv_mdlc_pc_virtual_district_set
				}
			}
			AND = {
				testmod_is_pd_hive_overmind_world = yes
				OR = {
					has_district = district_pd_fortress_arcology_housing
					has_district = district_pd_fortress_arcology_urban_1
					has_district = district_pd_fortress_arcology_urban_2
					has_district = district_pd_fortress_arcology_urban_3

				testmod_is_pd_hive_overmind_world = yes

					has_district = district_hive_1

					has_district = district_hive_3
				}
			}
		}
	}

		always = yes
	}

		while = {
			limit = {
				has_district = xvcv_mdlc_pc_virtual_district_primary
			}
			remove_district = xvcv_mdlc_pc_virtual_district_primary
		}

			}
			remove_district = district_pd_fortress_arcology_housing
			add_district = district_hive
		}
		while = {
			limit = {
				testmod_is_pd_hive_overmind_world = yes
				has_district = district_pd_fortress_arcology_urban_1
			}
			remove_district = district_pd_fortress_arcology_urban_1

		}

			limit = {
				testmod_is_pd_hive_overmind_world = yes
				has_district = district_pd_fortress_arcology_urban_2
			}
			remove_district = district_pd_fortress_arcology_urban_2
			add_district = district_hive
		}
		while = {
			limit = {
				testmod_is_pd_hive_overmind_world = yes
				has_district = district_pd_fortress_arcology_urban_3
			}
			remove_district = district_pd_fortress_arcology_urban_3
